四、网络协议（自定义二进制）

4.1 选型理由

对比维度JSON 文本自定义二进制
依赖需要 nlohmann/json零依赖，纯 C++ struct + memcpy
单条消息大小80-200 字节8-40 字节
解析开销字符串解析 + 内存分配直接强转 struct 指针
调试肉眼可读hexdump + 对照表，或写 to_string()
扩展性随意加字段新增消息类型即可，老类型不变

回合制总共 14 条消息，二进制协议手写不超过 200 行代码。

4.2 统一帧格式

偏移字节数字段说明
0x002msg_len整个帧的字节数（含自身），大端 uint16_t
0x022msg_type消息类型枚举，大端 uint16_t
0x041player_id发送者 ID（0=server, 1-6=玩家）
0x051flagsbit0=需要ACK，bit1=广播标记，其余保留
0x06Npayload各消息类型自定义（见 4.4）

4.3 msg_types.h

#pragma once
#include <cstdint>

enum class MsgType : uint16_t {
    // 房间阶段
    JOIN_REQ      = 0x0100,
    JOIN_ACK      = 0x0101,
    CLASS_OPT     = 0x0102,
    CLASS_PICK    = 0x0103,
    CLASS_CONF    = 0x0104,
    // 回合阶段
    TURN_START    = 0x0200,
    ACTION_REQ    = 0x0201,
    DICE_ROLL     = 0x0202,
    EVENT_FIRE    = 0x0203,
    TURN_END      = 0x0204,
    // 系统
    HEARTBEAT     = 0xFF00,
    HEARTBEAT_ACK = 0xFF01,
    PLAYER_DC     = 0xFF02,
    GAME_OVER     = 0xFF03,
};

// 统一帧头，6 字节
struct __attribute__((packed)) FrameHeader {
    uint16_t msg_len;    // 含头部的帧总长
    uint16_t msg_type;
    uint8_t  player_id;
    uint8_t  flags;
};
static_assert(sizeof(FrameHeader) == 6, "FrameHeader must be 6 bytes");

4.4 protocol.h 读写接口

// 发送端：把各消息的 Payload struct 包进帧头
template<typename Payload>
std::vector<uint8_t> pack(MsgType type, uint8_t player_id,
                           uint8_t flags, const Payload& payload) {
    uint16_t total = sizeof(FrameHeader) + sizeof(Payload);
    std::vector<uint8_t> buf(total);
    auto* hdr = reinterpret_cast<FrameHeader*>(buf.data());
    hdr->msg_len   = htons(total);
    hdr->msg_type  = htons(static_cast<uint16_t>(type));
    hdr->player_id = player_id;
    hdr->flags     = flags;
    std::memcpy(buf.data() + sizeof(FrameHeader), &payload, sizeof(Payload));
    return buf;
}

// 空 payload 消息 (HEARTBEAT 等)
inline std::vector<uint8_t> pack(MsgType type, uint8_t player_id, uint8_t flags) {
    uint16_t total = sizeof(FrameHeader);
    std::vector<uint8_t> buf(total);
    auto* hdr = reinterpret_cast<FrameHeader*>(buf.data());
    hdr->msg_len   = htons(total);
    hdr->msg_type  = htons(static_cast<uint16_t>(type));
    hdr->player_id = player_id;
    hdr->flags     = flags;
    return buf;
}

// 接收端：从帧头指针取出 payload
template<typename Payload>
const Payload& unpack_payload(const FrameHeader* hdr) {
    return *reinterpret_cast<const Payload*>(
        reinterpret_cast<const uint8_t*>(hdr) + sizeof(FrameHeader));
}

接收分帧：

void on_recv(int fd) {
    uint8_t len_buf[2];
    recv(fd, len_buf, 2, MSG_WAITALL);
    uint16_t total = ntohs(*reinterpret_cast<uint16_t*>(len_buf));

    std::vector<uint8_t> buf(total);
    std::memcpy(buf.data(), len_buf, 2);
    recv(fd, buf.data() + 2, total - 2, MSG_WAITALL);

    auto* hdr = reinterpret_cast<FrameHeader*>(buf.data());
    dispatch(hdr);  // 根据 msg_type 路由
}

房间阶段

消息Payload struct大小说明

消息Payload struct大小说明

ACTION_REQ { uint8_t action; uint8_t target; } 2B C→S，action=0探索,1攻击,2技能,3道具
DICE_ROLL { uint8_t p; uint8_t n; uint8_t d[6]; int16_t mod; int16_t total; } 12B S→广播

七、事件引擎（行为树驱动）

为什么逻辑层不需要 JSON：事件配置本质上就是一个阈值查表——给定骰子值，返回对应的事件 ID 和效果列表。这用 C++ constexpr 数组表达最直接，编译进二进制，零解析开销。BehaviorTree.CPP 的行为树结构（AI 决策树、事件分支树）仍然用 XML 文件——那是 BT 库的原生格式，不算额外依赖。

7.1 事件数据：纯 C++ constexpr

// event_data.h — 所有事件配置，编译时即确定，无需解析任何外部文件
#pragma once
#include <cstdint>

enum class EffectType : uint8_t { GOLD = 0, ITEM, COMBAT, DMG, HEAL, BUFF };

struct Effect {
    EffectType type;
    int16_t    value;       // 金币数 / 物品ID / 怪物ID / HP变化量 / buff ID
};

struct EventThreshold {
    int16_t  min_roll;      // 骰子最终值 >= 此值则触发
    uint16_t event_id;
    const char* desc;       // 仅用于 UI 展示，不参与逻辑
    Effect    effects[4];   // 最多 4 个效果
    uint8_t   effect_count;
};

struct EventTemplate {
    uint16_t            event_id;
    const char*         name;
    uint8_t             dice_count;
    uint8_t             dice_faces;
    EventThreshold      thresholds[4];  // 按 min_roll 降序排列
    uint8_t             threshold_count;
};

7.2 事件引擎查询

// event_engine.h
class EventEngine {
public:
    // 根据骰子最终值查找触发哪个事件
    const EventThreshold* resolve(uint16_t event_id, int16_t final_roll) const {
        const auto* tmpl = find_template(event_id);
        if (!tmpl) return nullptr;
        // thresholds 已按 min_roll 降序排列，找到第一个满足的
        for (int i = 0; i < tmpl->threshold_count; i++) {
            if (final_roll >= tmpl->thresholds[i].min_roll) {
                return &tmpl->thresholds[i];
            }
        }
        return nullptr;  // 不应到达，最后一个 threshold 的 min 总是 0
    }

    const EventTemplate* find_template(uint16_t event_id) const {
        for (size_t i = 0; i < EVENT_DB_SIZE; i++) {
            if (EVENT_DB[i].event_id == event_id) return &EVENT_DB[i];
        }
        return nullptr;
    }
};

7.3 行为树节点

事件分支连线仍是行为树的职责。以下是 BehaviorTree.CPP 的 XML + C++ Action 节点：

// TombExplore 行为树 Action 节点
class TombExplore : public BT::SyncActionNode {
public:
    BT::NodeStatus tick() override {
        auto& player  = config().blackboard->get<Player>("player");
        int roll      = config().blackboard->get<int>("dice_result");
        uint16_t evt  = config().blackboard->get<uint16_t>("event_id");

        EventEngine engine;
        const auto* threshold = engine.resolve(evt, roll);
        if (!threshold) return BT::NodeStatus::FAILURE;

        // 把匹配到的效果写入黑板，供子节点消费
        config().blackboard->set("matched_effects", threshold->effects);
        config().blackboard->set("matched_count",   threshold->effect_count);
        config().blackboard->set("event_desc",      threshold->desc);

        return BT::NodeStatus::SUCCESS;  // 继续走子行为树
    }
};

XML 连线（BehaviorTree.CPP 原生格式）：

      <Switch4 case_0="treasure" case_1="combat" case_2="trap" variable="matched_effects"/>

    SDL_Init(SDL_INIT_VIDEO | SDL_INIT_AUDIO);
    TTF_Init();

    auto window   = SDL_CreateWindow("Game", SDL_WINDOWPOS_CENTERED,
                                      SDL_WINDOWPOS_CENTERED, 1280, 720, 0);
    auto renderer = SDL_CreateRenderer(window, -1, SDL_RENDERER_ACCELERATED);

    ScreenManager screens;
    GameClient   client;

    // 解析命令行参数决定角色：主机 or 客户端
    if (is_host) {
        client.start_server(8888);
    }
    client.connect("192.168.43.1", 8888);

    screens.push(make_unique<TitleScreen>(client));

    bool running = true;
    Uint32 last_tick = SDL_GetTicks();
    while (running) {
        Uint32 now = SDL_GetTicks();
        float dt = (now - last_tick) / 1000.0f;
        last_tick = now;

        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) running = false;
            screens.current()->on_event(e);
        }

        client.poll();                        // 处理网络消息
        screens.current()->on_update(dt);
        SDL_RenderClear(renderer);
        screens.current()->on_render(renderer);
        SDL_RenderPresent(renderer);
    }

    SDL_Quit();
    return 0;
}

8.5 Sprite & SpriteAnim

// sprite.h
struct Sprite {
    SDL_Texture* tex = nullptr;
    SDL_Rect     src = {};        // 精灵表中的像素矩形
    SDL_Rect     dst = {};        // 屏幕位置 + 缩放后大小
    int          scale = 3;       // 16x16 → 48x48

    void draw(SDL_Renderer* r) {
        if (tex) SDL_RenderCopy(r, tex, &src, &dst);
    }

    static Sprite make(SDL_Texture* t, int src_x, int src_y,
                       int dst_x, int dst_y, int scale = 3) {
        return {t, {src_x, src_y, 16, 16},
                {dst_x, dst_y, 16 * scale, 16 * scale}, scale};
    }
};

// sprite_anim.h
class SpriteAnim {
    const Sprite* m_frames;
    int m_count, m_index = 0;
    float m_timer = 0, m_interval = 0.15f;
    bool m_loop = true, m_done = false;
public:
    SpriteAnim(const Sprite* frames, int count, float interval = 0.15f)
        : m_frames(frames), m_count(count), m_interval(interval) {}

    void update(float dt) {
        if (m_done) return;
        m_timer += dt;
        if (m_timer >= m_interval) {
            m_timer -= m_interval;
            if (++m_index >= m_count)
                m_loop ? m_index = 0 : (m_index = m_count - 1, m_done = true);
        }
    }

    void draw(SDL_Renderer* r) {
        Sprite s = m_frames[m_index];
        s.dst = m_frames[0].dst;   // 保持屏幕位置不变
        s.draw(r);
    }

    void set_loop(bool l) { m_loop = l; }
    bool is_done() const { return m_done; }
    void reset() { m_index = 0; m_timer = 0; m_done = false; }
};

8.6 Tilemap & Character

// tilemap.h — 瓦片地图
class Tilemap {
    SDL_Texture* m_tex;
    int m_cols, m_rows, m_scale = 3;
    vector<int> m_data;     // tile_id, 0 = 空
public:
    Tilemap(SDL_Texture* t, int cols, int rows) : m_tex(t), m_cols(cols), m_rows(rows), m_data(cols*rows) {}

    void set(int c, int r, int id) { m_data[r * m_cols + c] = id; }

    void draw(SDL_Renderer* r) {
        int ts = 16 * m_scale;
        SDL_Rect src = {0,0,16,16}, dst = {0,0,ts,ts};
        for (int y = 0; y < m_rows; y++)
            for (int x = 0; x < m_cols; x++) {
                int id = m_data[y * m_cols + x];
                if (!id--) continue;
                src.x = (id % 16) * 16; src.y = (id / 16) * 16;
                dst.x = x * ts; dst.y = y * ts;
                SDL_RenderCopy(r, m_tex, &src, &dst);
            }
    }
};

// character.h — 像素角色
enum class CharState { IDLE, WALK, ATTACK, HURT };

class Character {
    SpriteAnim *m_idle, *m_walk, *m_attack, *m_hurt, *m_current;
    CharState m_state = CharState::IDLE;
    int m_x, m_y, m_hp, m_max_hp;
    const char* m_name;
public:
    int y() const { return m_y; }  // 用于排序遮挡

    void set_state(CharState s) {
        if (m_state == s) return;
        m_state = s;
        switch (s) {
            case CharState::IDLE:   m_current = m_idle;   break;
            case CharState::WALK:   m_current = m_walk;   break;
            case CharState::ATTACK: m_current = m_attack; m_attack->set_loop(false); m_attack->reset(); break;
            case CharState::HURT:   m_current = m_hurt;   break;
        }
    }

    void update(float dt) { m_current->update(dt); }
    void draw(SDL_Renderer* r) { m_current->draw(r); draw_hp(r); draw_name(r); }
};

10.3 每条消息的端到端流程

以 ACTION_REQ → DICE_ROLL 为例：

序号执行方动作涉及模块
1客户端 UI玩家点击"探索"按钮GameScreen → Action 回调
2客户端 net序列化 ACTION_REQ，TCP 发送protocol::encode() → Client::send()
3服务端 net分帧读取，反序列化，路由到 TurnMgrServer::recv() → protocol::decode() → TurnMgr
4服务端逻辑TurnMgr 调用 DiceSystem.roll()，得到结果TurnMgr → DiceSystem
5服务端逻辑EventEngine 根据骰子结果查事件表TurnMgr → EventEngine (行为树 tick)
6服务端 netbroadcast DICE_ROLL + EVENT_FIREServer::broadcast()
7各客户端 net接收消息，回调 UIClient::poll() → GameScreen 回调
8各客户端 UI播放骰子动画 → 展示事件描述 → 更新 HPDiceAnim widget → GameScreen::on_update()
